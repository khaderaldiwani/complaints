═══════════════════════════════════════════════════════════════

┌─────────────────────────────────────────────────────────────┐
│  الخطوة 1: تثبيت المتطلبات                                  │
└─────────────────────────────────────────────────────────────┘

composer install


┌─────────────────────────────────────────────────────────────┐
│  الخطوة 2: إنشاء ملف .env                                   │
└─────────────────────────────────────────────────────────────┘

Windows:
copy .env.example .env

┌─────────────────────────────────────────────────────────────┐
│  الخطوة 3: تعديل ملف .env                                   │
│  (افتح الملف وعدّل إعدادات قاعدة البيانات)                 │
└─────────────────────────────────────────────────────────────┘

DB_DATABASE=complaints_db
DB_USERNAME=root
DB_PASSWORD=your_password


┌─────────────────────────────────────────────────────────────┐
│  الخطوة 4: إنشاء مفتاح التطبيق                             │
└─────────────────────────────────────────────────────────────┘

php artisan key:generate


┌─────────────────────────────────────────────────────────────┐
│  الخطوة 5: إنشاء قاعدة البيانات (من MySQL)                  │
└─────────────────────────────────────────────────────────────┘

mysql -u root -p

CREATE DATABASE complaints_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
EXIT;


┌─────────────────────────────────────────────────────────────┐
│  الخطوة 6: تشغيل Migrations                                  │
└─────────────────────────────────────────────────────────────┘

php artisan migrate


┌─────────────────────────────────────────────────────────────┐
│  الخطوة 7: مسح الكاش                                         │
└─────────────────────────────────────────────────────────────┘

php artisan config:clear
php artisan cache:clear
php artisan route:clear


┌─────────────────────────────────────────────────────────────┐
│  الخطوة 8: تشغيل السيرفر (نافذة Terminal 1)                │
└─────────────────────────────────────────────────────────────┘

php artisan serve


┌─────────────────────────────────────────────────────────────┐
│  الخطوة 9: تشغيل Queue Worker (نافذة Terminal 2) ⚠️ مهم!  │
└─────────────────────────────────────────────────────────────┘

php artisan queue:work

http://localhost:8000/api/agencies
